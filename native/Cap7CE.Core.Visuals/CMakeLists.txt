cmake_minimum_required(VERSION 3.24)
project(Cap7CEVisuals LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
add_compile_options(/utf-8)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(OPENEXR_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(OPENEXR_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(OPENEXR_INSTALL OFF CACHE BOOL "" FORCE)
set(OPENEXR_INSTALL_TOOLS OFF CACHE BOOL "" FORCE)
set(IMATH_INSTALL OFF CACHE BOOL "" FORCE)
set(IMATH_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(OPENEXR_FORCE_INTERNAL_IMATH ON CACHE BOOL "" FORCE)
set(OPENEXR_FORCE_INTERNAL_DEFLATE ON CACHE BOOL "" FORCE)
set(OPENEXR_FORCE_INTERNAL_OPENJPH ON CACHE BOOL "" FORCE)
set(FETCHCONTENT_SOURCE_DIR_IMATH "${CAP_IMATH_SOURCE}" CACHE PATH "" FORCE)
add_subdirectory("${CAP_OPENEXR_SOURCE}" openexr EXCLUDE_FROM_ALL)
add_executable(Cap7CE.Core.Visuals main.cpp job_runner.cpp job_control.cpp scratch.cpp thumbnail.cpp
  icon_pixels.cpp pdf_thumbnail.cpp exr_source.cpp exr_render.cpp)
target_compile_definitions(Cap7CE.Core.Visuals PRIVATE UNICODE _UNICODE NOMINMAX _WIN32_WINNT=0x0A00)
target_compile_options(Cap7CE.Core.Visuals PRIVATE /W4 /WX /utf-8 /EHsc /external:anglebrackets /external:W0)
target_link_libraries(Cap7CE.Core.Visuals PRIVATE OpenEXR::OpenEXR windowsapp ole32 shell32 windowscodecs uuid gdi32 user32)
target_link_options(Cap7CE.Core.Visuals PRIVATE "${CAP_RESOURCE}" /MANIFEST:EMBED "/MANIFESTUAC:level='asInvoker' uiAccess='false'")
set_property(TARGET Cap7CE.Core.Visuals APPEND PROPERTY LINK_DEPENDS "${CAP_RESOURCE}")
set_target_properties(Cap7CE.Core.Visuals PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CAP_OUTPUT}")
